// Kensei Weapon Override

IF
LeveledUp(_Character)
AND
IsTagged(_Character,MONK_5EKENSEI_3adb6cd0-1543-430f-bccf-8aaf2df0dab0,1)
THEN
IterateInventory(_Character,"CheckKenseiWeapons","CompleteKenseiWeaponCheck");

// Override Kensei Weapon on Inventory Check Part 1

IF
EntityEvent((ITEM)_Item,"CheckKenseiWeapons")
AND
IsWeapon(_Item,1)
AND
GetInventoryOwner(_Item,_Character)
AND
DB_KenseiWeaponsInfo(_Status,_Tag)
AND
IsTagged(_Item,_Tag,1)
AND
HasActiveStatus(_Character,_Status,1)
THEN
SetDualEntityEvent(_Character, _Item, "KenseiWeaponDamageDie", 0);

// Override Kensei Weapon on Inventory Check Part 2

AND
HasActiveStatus(_Character,_Status,1)

AND
IsTagged(_Item,_Tag,1)
THEN
ApplyStatus(_Item,"KENSEI_WEAPON_OVERRIDE_D6",-1.0,1,_Character);
